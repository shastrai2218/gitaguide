# ---------------------------------------------------------------------------
# GitaGuide SERVING runtime (Python 3.11)
#
# This file is the production/serving dependency set and it is deliberately
# **torch-free**. Installing it must not pull PyTorch: the emotion model and the
# embedding model both run as ONNX graphs on onnxruntime, so a training
# framework is dead weight in a serving container (torch alone is ~4.4 GB
# installed; see docs/kg_measurements.md section 9).
#
# Anything that needs torch - the sentence-transformers parity check, dev tools,
# linting - lives in requirements-dev.txt.
#
# With GITAGUIDE_MOCK_MODE=true the app runs on in-process fakes, so the
# "offline core" block alone is enough to run the API, the tests and the eval
# harness.
#
# Versions are pinned to the stack every measurement in docs/ was taken on.
# ---------------------------------------------------------------------------

# ---- offline core (required) ----------------------------------------------

pydantic-settings==2.14.2

# ---- conversational router ------------------------------------------------
# Reads data/intent_lexicon.yaml only. A data file rather than a Python literal
# so the greeting/acknowledgement terms can grow - in four languages - without
# a code change or a test author's involvement. ~800 KB installed, no
# transitive dependencies, and only `safe_load` is ever called.
PyYAML==6.0.3

# ---- corpus acquisition (scripts/download_arthashastra.py) ----------------
requests==2.34.2

# ---- persistence ----------------------------------------------------------

asyncpg==0.31.0            # PostgreSQL async driver
aiosqlite==0.22.1          # SQLite async driver (tests / laptop mode)

# ---- auth -------------------------------------------------------------
# Verifies Clerk-issued session JWTs against its published JWKS (RS256), so
# accounts are optional identity on top of the existing anonymous sessions,
# not a new service this process talks to. `[crypto]` pulls in `cryptography`
# for RS256; there is no lighter way to check an RSA signature.
PyJWT[crypto]==2.11.0

# ---- models: both run on onnxruntime, CPU, no torch -----------------------
# emotion   : SamLowe/roberta-base-go_emotions, ONNX INT8
# embeddings: sentence-transformers/all-MiniLM-L6-v2, ONNX fp32
# `transformers` is here for AutoTokenizer/AutoConfig only. It does not pull
# torch by itself; tests/test_torch_free_runtime.py holds that line.
transformers==5.15.1
tokenizers==0.22.2
onnxruntime==1.29.0
huggingface-hub==1.28.0
numpy==2.4.4

neo4j-graphrag==1.18.0

# ---- LLM providers -------------------------------------------------------
